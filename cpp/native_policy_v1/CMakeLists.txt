cmake_minimum_required(VERSION 3.16)
project(weighted_grpo_native_policy_v1 LANGUAGES CXX)
add_library(weighted_grpo_native_policy_v1 STATIC policy.cpp)
target_compile_features(weighted_grpo_native_policy_v1 PUBLIC cxx_std_20)
target_include_directories(weighted_grpo_native_policy_v1 PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
set_target_properties(weighted_grpo_native_policy_v1 PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(weighted_grpo_native_policy_v1 PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math -ffp-contract=off)
elseif(MSVC)
  target_compile_options(weighted_grpo_native_policy_v1 PRIVATE /W4 /fp:strict)
endif()
add_executable(weighted_grpo_native_policy_cli main.cpp)
target_link_libraries(weighted_grpo_native_policy_cli PRIVATE weighted_grpo_native_policy_v1)
include(CTest)
if(BUILD_TESTING)
  add_executable(native_policy_sha256_test test_sha256.cpp)
  target_link_libraries(native_policy_sha256_test PRIVATE weighted_grpo_native_policy_v1)
  add_test(NAME native_policy_sha256_known_answers COMMAND native_policy_sha256_test)
  add_executable(native_policy_greedy_contract greedy_contract.cpp)
  target_link_libraries(native_policy_greedy_contract PRIVATE weighted_grpo_native_policy_v1)
  add_test(NAME native_policy_greedy_synthetic_contract COMMAND native_policy_greedy_contract)
  set(NATIVE_POLICY_TEST_MODEL "" CACHE FILEPATH "Optional real WGPN model for greedy parity and microbenchmark")
  if(NATIVE_POLICY_TEST_MODEL)
    add_test(NAME native_policy_greedy_real_model
             COMMAND native_policy_greedy_contract ${NATIVE_POLICY_TEST_MODEL})
  endif()
endif()
