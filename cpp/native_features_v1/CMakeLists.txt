cmake_minimum_required(VERSION 3.18)
project(wc_native_features_v1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(wc_native_features_v1 STATIC features.cpp)
target_include_directories(wc_native_features_v1 PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(wc_native_features_v1 PRIVATE -fno-fast-math -ffp-contract=off)
endif()
add_executable(native_features_cli features_cli.cpp)
target_link_libraries(native_features_cli PRIVATE wc_native_features_v1)
